Auth
POST   /api/auth/register
POST   /api/auth/login
GET    /api/auth/user
PUT    /api/auth/user
DELETE /api/auth/user

Shop
POST   /api/shops              # create shop
GET    /api/shops/{shopId}     # view shop details
GET    /api/shops              # get shops by owner id
PUT    /api/shops              # update shop name
DELETE /api/shops              # delete shop

Menu
POST   /api/menu                  # add menu
POST   /api/menu/{menuId}         # set menu
GET    /api/menus/{shopId}        # list menus 
DELETE /api/menu/{menuId}         # delete menu

Items
POST   /api/item             # add item
GET    /api/item/{itemId}    # list item
GET    /api/item/{shopId}    # list items
PUT    /api/item/{itemId}    # update item
DELETE /api/item/{itemId}    # delete item

Menu Items
POST   /api/menu-item             # add menu item (itemid and menuid in the req body)
GET    /api/menu-item/{menuId}    # list menu items
DELETE /api/menu-item             # delete menu item   (itemid and menuid in the req body)
DELETE /api/menu-item/{menuId}    # delete menu items

Category
POST   /api/category            # add category
GET    /api/category            # list categories
DELETE /api/category            # delete category

Table
POST   /api/tables                     # create table + generate QR
GET    /api/tables/{shopId}            # list tables
GET    /api/tables/{tableId}/session   # get current active session (call this before order, if doesnt exist, then create)

Session
POST   /api/sessions                    # start session (tableId)
GET    /api/sessions/{sessionId}        # get session info
PUT    /api/sessions/{sessionId}/close

Order (order then order_items)
POST   /api/orders/{tableId}         # create order (auto-create session if none exists)
GET    /api/orders/{tableId}         # list all orders for the current active session
PUT    /api/orders/{orderId}/status             # update order (accept/reject/complete)

Payments
POST   /api/sessions/{sessionId}/payments       # record payment
GET    /api/sessions/{sessionId}/payments       # check payment status
